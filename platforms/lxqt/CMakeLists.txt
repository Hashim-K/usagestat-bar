cmake_minimum_required(VERSION 3.16)
project(usagestat_lxqt LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets DBus)
find_path(LXQT_PANEL_INCLUDE_DIR ilxqtpanelplugin.h PATH_SUFFIXES lxqt lxqt-panel REQUIRED)
add_library(usagestat MODULE plugin.cpp)
target_include_directories(usagestat PRIVATE ${LXQT_PANEL_INCLUDE_DIR})
target_link_libraries(usagestat PRIVATE Qt6::Widgets Qt6::DBus)
target_compile_options(usagestat PRIVATE -Wall -Wextra -Werror)
